### summary function
data_summary <- function(data, varname, groupnames){
  require(plyr)
  summary_func <- function(x, col){
    c(mean = mean(x[[col]], na.rm=TRUE),
      se = sd(x[[col]], na.rm=TRUE) / sqrt(length(x[[col]])))
  }
  data_sum<-ddply(data, groupnames, .fun=summary_func,
                  varname)
  data_sum <- plyr::rename(data_sum, c("mean" = varname))
  return(data_sum)
}

    cat("K-means clustering results (k =", optimal_k, "):\n")
    print(km$size)
    
    # 5. Plot clusters
    fviz_cluster(km, data = vignette_scaled, main = paste("K-Means Clusters - Vignette", v))
    
    # === NEW: store cluster assignments ===
    cluster_assignments[[as.character(v)]] <- data.frame(
      PROLIFIC_PID = pids,
      Vignette = v,
      Cluster = km$cluster
    )
    
  } else {
    cat("Not enough complete responses for Vignette", v, "\n")
  }
}
